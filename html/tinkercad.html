Desafio 1: Iluminação Inteligente

Sistema automatizado que aciona um LED apenas quando há presença de pessoas E o ambiente encontra-se escuro, utilizando um sensor PIR e um LDR.

1. Circuito no Tinkercad

2. Código do Arduino


const int pinPIR = 2;
const int pinLDR = A0;
const int pinLED = 13;

int limiarLuminosidade = 400; // Ajuste conforme a simulação do Tinkercad

void setup() {
  pinMode(pinPIR, INPUT);
  pinMode(pinLED, OUTPUT);
  pinMode(pinLDR, INPUT);
  Serial.begin(9600);
}

void loop() {
  int presenca = digitalRead(pinPIR);
  int leituraLuminosidade = analogRead(pinLDR);

  // Lógica: Pessoa presente (HIGH) E Ambiente Escuro (leitura < limiar)
  if (presenca == HIGH && leituraLuminosidade < limiarLuminosidade) {
    digitalWrite(pinLED, HIGH);
  } else {
    digitalWrite(pinLED, LOW);
  }
  delay(100);
}

Desafio 2: Estacionamento Inteligente

Alerta visual de estacionamento utilizando sensor ultrassônico e potenciômetro. O potenciômetro define dinamicamente a distância de segurança, alterando a frequência de intermitência do LED.

1. Circuito no Tinkercad

2. Código do Arduino


const int pinTrig = 9;
const int pinEcho = 8;
const int pinLED = 11;
const int pinPot = A0;

long duracao;
float distancia;
int limiteSeguranca;

void setup() {
  pinMode(pinTrig, OUTPUT);
  pinMode(pinEcho, INPUT);
  pinMode(pinLED, OUTPUT);
  pinMode(pinPot, INPUT);
  Serial.begin(9600);
}

void loop() {
  // Ler potenciômetro e mapear para uma distância limite (ex: 10cm a 50cm)
  limiteSeguranca = map(analogRead(pinPot), 0, 1023, 10, 50);

  // Mapear pulso do sensor ultrassônico
  digitalWrite(pinTrig, LOW);
  delayMicroseconds(2);
  digitalWrite(pinTrig, HIGH);
  delayMicroseconds(10);
  digitalWrite(pinTrig, LOW);

  duracao = pulseIn(pinEcho, HIGH);
  distancia = duracao * 0.034 / 2;

  // Lógica de alerta proporcional à distância
  if (distancia < (limiteSeguranca / 2)) {
    // Muito próximo: LED aceso direto
    digitalWrite(pinLED, HIGH);
  } 
  else if (distancia <= limiteSeguranca) {
    // Se aproximando: LED pisca rápido conforme se aproxima
    int tempoPisca = map(distancia, limiteSeguranca / 2, limiteSeguranca, 50, 400);
    digitalWrite(pinLED, HIGH);
    delay(tempoPisca);
    digitalWrite(pinLED, LOW);
    delay(tempoPisca);
  } 
  else {
    // Fora do limite: LED desligado
    digitalWrite(pinLED, LOW);
  }
}

Desafio 3: Ambiente Inteligente (Estufa)

Monitoramento climático de estufa utilizando sensor de temperatura (TMP36) e sensor de umidade. Indicações visuais simultâneas via LEDs de estado.

1. Circuito no Tinkercad

2. Código do Arduino


const int pinTemp = A0;
const int pinUmidade = A1;

const int ledVerde = 2;   // Condição Ideal
const int ledVermelho = 3; // Temperatura Elevada
const int ledAmarelo = 4;  // Umidade Inadequada

const float limiteTempGrave = 30.0; // °C
const int limiteUmidadeMinima = 40;  // %

void setup() {
  pinMode(ledVerde, OUTPUT);
  pinMode(ledVermelho, OUTPUT);
  pinMode(ledAmarelo, OUTPUT);
  Serial.begin(9600);
}

void loop() {
  // Leitura do Sensor de Temperatura TMP36
  int leituraTemp = analogRead(pinTemp);
  float voltagem = leituraTemp * (5.0 / 1023.0);
  float temperatura = (voltagem - 0.5) * 100;

  // Leitura analógica do Sensor de Umidade
  int leituraUmidade = analogRead(pinUmidade);
  int umidadePercent = map(leituraUmidade, 0, 1023, 0, 100);

  bool tempAlta = (temperatura > limiteTempGrave);
  bool umidadeBaixa = (umidadePercent < limiteUmidadeMinima);

  // Apagar leds para atualização
  digitalWrite(ledVerde, LOW);
  digitalWrite(ledVermelho, LOW);
  digitalWrite(ledAmarelo, LOW);

  if (!tempAlta && !umidadeBaixa) {
    // Condição Adequada
    digitalWrite(ledVerde, HIGH);
  } else {
    // Condições Atípicas (suporta acionamento simultâneo)
    if (tempAlta) digitalWrite(ledVermelho, HIGH);
    if (umidadeBaixa) digitalWrite(ledAmarelo, HIGH);
  }

  delay(200);
}
